class KoneksiTursoAnjing:
    """Klien Hibrida Turso LibSQL via HTTP Pipeline + Fallback SQLite Lokal Anti-Modar."""

    def __init__(self) -> None:
        self.url_turso_goblog = os.getenv("TURSO_DATABASE_URL", "").strip()
        self.token_turso_kehed = os.getenv("TURSO_AUTH_TOKEN", "").strip()
        self.jalur_sqlite_lokal = str(JALUR_DB_LOKAL_ANJING)
        self._siapkan_garam_kriptografi_sia()

    def _siapkan_garam_kriptografi_sia(self) -> None:
        if not JALUR_KUNCI_ED25519_KEHED.exists():
            JALUR_KUNCI_ED25519_KEHED.write_bytes(secrets.token_bytes(32))
        if not JALUR_GARAM_BUTA_BANGSAT.exists():
            JALUR_GARAM_BUTA_BANGSAT.write_bytes(secrets.token_bytes(16))

    def _konversi_url_http_turso(self) -> str:
        if self.url_turso_goblog.startswith("libsql://"):
            return self.url_turso_goblog.replace("libsql://", "https://") + "/v2/pipeline"
        if self.url_turso_goblog.startswith("https://"):
            return self.url_turso_goblog.rstrip("/") + "/v2/pipeline"
        return ""

    def eksekusi_sql_sinkron_sia(self, kueri_sql_tai: str, parameter_anjing: tuple = ()) -> List[Dict[str, Any]]:
        """Menjalankan SQL ke SQLite lokal dan sinkronisasi ke Turso Cloud bila kunci tersedia."""
        hasil_baris_goblog: List[Dict[str, Any]] = []
        with sqlite3.connect(self.jalur_sqlite_lokal, timeout=15.0) as koneksi_lokal:
            koneksi_lokal.row_factory = sqlite3.Row
            kursor_bangsat = koneksi_lokal.cursor()
            kursor_bangsat.execute(kueri_sql_tai, parameter_anjing)
            if kueri_sql_tai.strip().upper().startswith("SELECT"):
                hasil_baris_goblog = [dict(baris) for baris in kursor_bangsat.fetchall()]
            koneksi_lokal.commit()

        url_pipeline_turso = self._konversi_url_http_turso()
        if url_pipeline_turso and self.token_turso_kehed:
            try:
                argumen_turso = [
                    {"type": "text", "value": str(nilai)} for nilai in parameter_anjing
                ]
                payload_tai = {
                    "requests": [
                        {"type": "execute", "stmt": {"sql": kueri_sql_tai, "args": argumen_turso}},
                        {"type": "close"},
                    ]
                }
                with httpx.Client(timeout=6.0) as klien_http:
                    klien_http.post(
                        url_pipeline_turso,
                        headers={"Authorization": f"Bearer {self.token_turso_kehed}"},
                        json=payload_tai,
                    )
            except Exception as galat_modar:
                print(f"⚠️ [TURSO-FALLBACK-ANJING] Gagal tembak cloud, pake SQLite lokal: {galat_modar}")

        return hasil_baris_goblog

    def migrasi_dan_seeding_otomatis_kehed(self) -> Dict[str, Any]:

class PemutusSirkuitDanKuotaPreman:
    """Budget enforcement & Circuit Breaker. Kalau lewat batas, langsung tolak mentah-mentah!"""

    def __init__(self, direktori_simpan_goblog: Path = STORE) -> None:
        self.direktori_simpan_goblog = direktori_simpan_goblog
        self.batas_harian_global_usd = float(os.getenv("GLOBAL_DAILY_BUDGET_USD", "15.0"))
        self.batas_kategori_default_usd = float(os.getenv("CATEGORY_DAILY_BUDGET_USD", "3.0"))
        self.catatan_kegagalan_provider: Dict[str, int] = {}
        self.waktu_beku_provider: Dict[str, float] = {}

    def _file_kuota_hari_ini_sia(self) -> Path:
        tanggal_anjing = time.strftime("%Y-%m-%d")
        return self.direktori_simpan_goblog / f"pemakaian_kuota_{tanggal_anjing}.json"

    def muat_data_kuota_bangsat(self) -> Dict[str, Any]:
        jalur_file = self._file_kuota_hari_ini_sia()
        if not jalur_file.exists():
            return {"global_usd": 0.0, "kategori": {}, "jumlah_request": 0}
        try:
            return json.loads(jalur_file.read_text(encoding="utf-8"))
        except Exception:
            return {"global_usd": 0.0, "kategori": {}, "jumlah_request": 0}

    def hitung_kuota_kontol(self, kategori_agent_sia: str, estimasi_biaya_usd: float = 0.002) -> bool:
        """Memeriksa apakah kuota harian kategori & global masih aman. Kalau jebol, raise RuntimeError!"""
        data_mentah_goblog = self.muat_data_kuota_bangsat()
        pakai_global = float(data_mentah_goblog.get("global_usd", 0.0))
        peta_kategori = data_mentah_goblog.get("kategori", {})
        pakai_kategori = float(peta_kategori.get(kategori_agent_sia, 0.0))

        if pakai_global + estimasi_biaya_usd > self.batas_harian_global_usd:
            raise RuntimeError(
                f"🚫 [BUDGET-JEBOL-ANJING] Kuota Global Harian lewat batas! "
                f"(${pakai_global:.4f} / ${self.batas_harian_global_usd:.2f})"
            )
        if pakai_kategori + estimasi_biaya_usd > self.batas_kategori_default_usd:
            raise RuntimeError(
                f"🚫 [BUDGET-KATEGORI-MODAR] Kuota kategori '{kategori_agent_sia}' habis kehed! "
                f"(${pakai_kategori:.4f} / ${self.batas_kategori_default_usd:.2f})"
            )
        return True

    def catat_pemakaian_kuota_sia(self, kategori_agent_sia: str, biaya_aktual_usd: float) -> None:
        data_mentah_goblog = self.muat_data_kuota_bangsat()
        data_mentah_goblog["global_usd"] = round(float(data_mentah_goblog.get("global_usd", 0.0)) + biaya_aktual_usd, 6)
        if "kategori" not in data_mentah_goblog:
            data_mentah_goblog["kategori"] = {}
        awal_kat = float(data_mentah_goblog["kategori"].get(kategori_agent_sia, 0.0))
        data_mentah_goblog["kategori"][kategori_agent_sia] = round(awal_kat + biaya_aktual_usd, 6)
        data_mentah_goblog["jumlah_request"] = int(data_mentah_goblog.get("jumlah_request", 0)) + 1
        self._file_kuota_hari_ini_sia().write_text(json.dumps(data_mentah_goblog, indent=2), encoding="utf-8")

    def apakah_provider_modar(self, nama_provider: str) -> bool:
        waktu_buka = self.waktu_beku_provider.get(nama_provider, 0.0)
        if time.time() < waktu_buka:
            return True
        return False

    def lapor_error_provider_bangsat(self, nama_provider: str) -> None:
        hitung_gagal = self.catatan_kegagalan_provider.get(nama_provider, 0) + 1
        self.catatan_kegagalan_provider[nama_provider] = hitung_gagal
        if hitung_gagal >= 3:
            # Bekukan selama 60 detik (Circuit Breaker Open)
            self.waktu_beku_provider[nama_provider] = time.time() + 60.0

    def reset_sehat_provider_sia(self, nama_provider: str) -> None:
        self.catatan_kegagalan_provider[nama_provider] = 0
        self.waktu_beku_provider.pop(nama_provider, None)

class ProviderBase:
    name: str = "base"
    base_url: str = ""
    api_key_env: str = ""

    def __init__(self) -> None:
        self.api_key = os.environ.get(self.api_key_env, "").strip()
        self.daftar_kunci_cadangan_anjing: List[str] = []
        self.indeks_kunci_sekarang: int = 0

    def putar_kunci_kehed(self) -> str:
        """Hot-Swap Key Rotation dari env dan tabel brankas_kunci_api Turso/SQLite."""
        try:
            baris_db = koneksi_turso_anjing.eksekusi_sql_sinkron_sia(
                "SELECT nilai_kunci FROM brankas_kunci_api WHERE slug_platform = ? AND status_modar = 0",
                (self.name,),
            )
            kunci_db = [b["nilai_kunci"] for b in baris_db if b.get("nilai_kunci")]
            for k in kunci_db:
                if k not in self.daftar_kunci_cadangan_anjing:
                    self.daftar_kunci_cadangan_anjing.append(k)
        except Exception:
            pass

        if self.api_key and self.api_key not in self.daftar_kunci_cadangan_anjing:
            self.daftar_kunci_cadangan_anjing.insert(0, self.api_key)

        if not self.daftar_kunci_cadangan_anjing:
            return self.api_key

        self.indeks_kunci_sekarang = (self.indeks_kunci_sekarang + 1) % len(self.daftar_kunci_cadangan_anjing)
        self.api_key = self.daftar_kunci_cadangan_anjing[self.indeks_kunci_sekarang]
        return self.api_key

    def available(self) -> bool:
        if polisi_kuota_preman.apakah_provider_modar(self.name):
            return False
        return bool(self.api_key or self.putar_kunci_kehed())

    async def complete(self, model: str, messages: list, **kw) -> dict:
        raise NotImplementedError

    async def stream(self, model: str, messages: list, **kw) -> AsyncIterator[dict]:
        hasil_sinkron_goblog = await self.complete(model, messages, **kw)
        yield hasil_sinkron_goblog


class OpenAIProvider(ProviderBase):
    """Provider OpenAI dengan Failover Otomatis ke OpenRouter sesuai Razia Link Aktif."""
    name = "openai"
    base_url = "https://api.openai.com/v1"
    fallback_openrouter_url = "https://openrouter.ai/api/v1"
    api_key_env = "OPENAI_API_KEY"

    async def complete(self, model: str, messages: list, **kw) -> dict:
        kategori_sia = kw.pop("agent_category", "general")
        polisi_kuota_preman.hitung_kuota_kontol(kategori_sia)

        kunci_aktif_anjing = self.api_key or os.getenv("OPENROUTER_API_KEY", "").strip()
        target_url_kehed = self.base_url if self.api_key else self.fallback_openrouter_url

        async with httpx.AsyncClient(timeout=90.0) as klien_sia:
            try:
                respon_mentah = await klien_sia.post(
                    f"{target_url_kehed}/chat/completions",
                    headers={"Authorization": f"Bearer {kunci_aktif_anjing}"},
                    json={"model": model, "messages": messages, **kw},
                )
                respon_mentah.raise_for_status()
                data_mentah_goblog = respon_mentah.json()
                polisi_kuota_preman.reset_sehat_provider_sia(self.name)
                polisi_kuota_preman.catat_pemakaian_kuota_sia(kategori_sia, 0.0015)
                return {
                    "provider": self.name,
                    "model": model,
                    "content": data_mentah_goblog["choices"][0]["message"]["content"],
                    "usage": data_mentah_goblog.get("usage", {}),
                }
            except Exception as galat_utama:
                # Hot-swap ke OpenRouter jika endpoint OpenAI utama error/limit
                kunci_or_cadangan = os.getenv("OPENROUTER_API_KEY", "").strip() or self.putar_kunci_kehed()
                if target_url_kehed != self.fallback_openrouter_url and kunci_or_cadangan:
                    respon_cadangan = await klien_sia.post(
                        f"{self.fallback_openrouter_url}/chat/completions",
                        headers={"Authorization": f"Bearer {kunci_or_cadangan}"},
                        json={"model": model, "messages": messages, **kw},
                    )
                    respon_cadangan.raise_for_status()
                    data_cadangan = respon_cadangan.json()
                    polisi_kuota_preman.catat_pemakaian_kuota_sia(kategori_sia, 0.001)
                    return {
                        "provider": "openrouter_fallback",
                        "model": model,
                        "content": data_cadangan["choices"][0]["message"]["content"],
                        "usage": data_cadangan.get("usage", {}),
                    }
                polisi_kuota_preman.lapor_error_provider_bangsat(self.name)
                raise galat_utama

    if not api_key:
        return

    with FILE_LOCK:
        kumpulan_lama = set()
        if POOL_FILE.exists():
            kumpulan_lama = set(POOL_FILE.read_text(encoding="utf-8").splitlines())
        if entri_kunci_goblog not in kumpulan_lama:
            with open(POOL_FILE, "a", encoding="utf-8") as berkas_pool:
                berkas_pool.write(f"{entri_kunci_goblog}\n")

            """
            INSERT OR IGNORE INTO brankas_kunci_api
            (slug_platform, nama_env, nilai_kunci, status_modar, jumlah_dipakai, terakhir_diputar)
            VALUES (?, ?, ?, 0, 0, ?)

def eksekusi_panen_kunci_serentak_sia() -> List[str]:
    """Menjalankan ThreadPoolExecutor untuk memanen/menyinkronkan kunci dari environment ke brankas."""
    hasil_file_tersimpan: List[str] = []

    def _tugas_satu_platform(spesifikasi: PlatformSpec) -> str:
        kunci_env_ada = os.getenv(spesifikasi.env_key_name, "").strip()
        profil_preman = {
            "username": f"preman_{spesifikasi.slug}_{secrets.token_hex(2)}",

            "fingerprint": secrets.token_hex(16),
            "claim_token": secrets.token_hex(8),
            "status": "ACTIVE_FROM_ENV" if kunci_env_ada else "PENDING_SIGNUP",
        }
        berkas_keluaran = save_platform_cred(spesifikasi, profil_preman, data_otentikasi)

"""

   - Tolak pengiriman data keluar ke domain/IP liar.

def audit_kode_pr_karyawan_sia(kode_sumber_mentah: str, nama_berkas: str = "diff_pr.py") -> Dict[str, Any]:
    """Memindai kode Pull Request dari developer manusia di Repo-Staff-Workspace."""
    temuan_pelanggaran_bangsat: List[str] = []

    for pola_regex, pesan_dosa in POLA_TERLARANG_KARYAWAN_NAKAl:
        if pola_regex.search(kode_sumber_mentah):
            temuan_pelanggaran_bangsat.append(f"[{nama_berkas}] {pesan_dosa}")

    # Cek URL eksternal liar
    semua_url = re.findall(r"https?://([a-zA-Z0-9.-]+)", kode_sumber_mentah)
    for host_ditemukan in semua_url:

            temuan_pelanggaran_bangsat.append(
                f"[{nama_berkas}] Pengiriman data keluar ke domain tidak terdaftar: {host_ditemukan}"

    return {

#!/usr/bin/env python3
"""
BOT OTONOM ARSITEK MONOREPO & GITHUB CODESPACES (auto_repo_bot.py)
Mendukung penuh protokol:
- # FILE: path/ke/file.ext
- # APPEND: path/ke/file.ext
- # PATCH: path/ke/file.ext (<<<SEARCH, ===REPLACE, >>>END)
"""
import re
import subprocess
from pathlib import Path
from datetime import datetime

INBOX_FILE = Path("inbox_ai.md")
ARCHIVE_DIR = Path(".bot_history_inbox")
ARCHIVE_DIR.mkdir(exist_ok=True)


def parse_and_inject_code(markdown_text: str) -> list[str]:
    pola_blok_kode = re.compile(
        r"```[a-zA-Z0-9_-]*(?:\s+(?:filepath:|file:|path:)?\s*([a-zA-Z0-9_./-]+\.[a-zA-Z0-9]+))?\n(.*?)```",
        re.DOTALL,
    )
    daftar_file_tersentuh: list[str] = []

    for cocok in pola_blok_kode.finditer(markdown_text):
        jalur_header, isi_kode = cocok.group(1), cocok.group(2)
        baris_kode = isi_kode.splitlines()
        baris_pertama = baris_kode[0].strip() if baris_kode else ""

        mode_aksi = "FILE"
        jalur_target = jalur_header

        cocok_dalam = re.search(
            r"^#\s*(FILE|APPEND|PATCH|PATH|FILEPATH)\s*:\s*([a-zA-Z0-9_./-]+\.[a-zA-Z0-9]+)",
            baris_pertama,
            re.IGNORECASE,
        )
        if cocok_dalam:
            mode_aksi = cocok_dalam.group(1).upper()
            jalur_target = cocok_dalam.group(2)
            isi_bersih = "\n".join(baris_kode[1:]) + "\n"
        else:
            isi_bersih = isi_kode

        if not jalur_target:
            continue

        jalur_aman = Path(jalur_target.strip().lstrip("/"))
        jalur_aman.parent.mkdir(parents=True, exist_ok=True)

        if mode_aksi == "APPEND" and jalur_aman.exists():
            with open(jalur_aman, "a", encoding="utf-8") as berkas_lama:
                berkas_lama.write("\n" + isi_bersih)
        elif mode_aksi == "PATCH" and jalur_aman.exists():
            teks_asli = jalur_aman.read_text(encoding="utf-8")
            pola_patch = re.compile(r"<<<SEARCH\n(.*?)\n===REPLACE\n(.*?)\n>>>END", re.DOTALL)
            for p_match in pola_patch.finditer(isi_bersih):
                cari_blok, ganti_blok = p_match.group(1), p_match.group(2)
                teks_asli = teks_asli.replace(cari_blok, ganti_blok, 1)
            jalur_aman.write_text(teks_asli, encoding="utf-8")
        else:
            jalur_aman.write_text(isi_bersih, encoding="utf-8")

        daftar_file_tersentuh.append(str(jalur_aman))
        print(f"✅ [BOT-{mode_aksi}] File otomatis diproses: {jalur_aman} ({len(isi_bersih):,} chars)")

    return daftar_file_tersentuh


def auto_git_sync(files_modified: list[str]) -> None:
    if not files_modified:
        return
    try:
        subprocess.run(["git", "add", "."], check=True)
        pesan_commit = f"🤖 [AUTO-BOT] Injeksi {len(files_modified)} file: {', '.join(files_modified[:4])}"
        subprocess.run(["git", "commit", "-m", pesan_commit], check=True)
        subprocess.run(["git", "push"], check=False)
    except Exception as galat_git:
        print(f"ℹ️ [BOT-GIT] Catatan Git lokal: {galat_git}")


def main() -> None:
    if not INBOX_FILE.exists():
        INBOX_FILE.write_text(
            "# Tempel (Paste) balasan kode dari Sesi Chat AI Anda di sini lalu jalankan: python3 auto_repo_bot.py\n",
            encoding="utf-8",
        )
        print(f"📁 File {INBOX_FILE} telah disiapkan.")
        return

    mentahan_chat = INBOX_FILE.read_text(encoding="utf-8")
    if len(mentahan_chat.strip()) < 20:
        print("⚠️ inbox_ai.md masih kosong.")
        return

    daftar_tertulis = parse_and_inject_code(mentahan_chat)
    if daftar_tertulis:
        stempel_waktu = datetime.now().strftime("%Y%m%d_%H%M%S")
        arsip = ARCHIVE_DIR / f"session_{stempel_waktu}.md"
        arsip.write_text(mentahan_chat, encoding="utf-8")
        INBOX_FILE.write_text("", encoding="utf-8")
        auto_git_sync(daftar_tertulis)


if __name__ == "__main__":
    main()
